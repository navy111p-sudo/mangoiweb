2-2. 사무실 모드 무음 수리 — 사람 승인 필요

  • 문제: 교사 화면은 「사무실 모드」(노이즈 게이트, js/idx-vc-officemode.js)가 기본 켜짐. 수업 중 AudioContext 가 running 이 아니게 되면 가공 트랙이 무음인데 트랙은 live 그대로라 마이크 자가치유도 안 돌고 안내도 없음. 이번 사고의 원인인지는 미확인(가능성 있는 자리).
  • 수리: tick() 에서 엔진이 멈추면 깨워 보고, 약 1초(40틱) 안에 안 살아나면 disable(true) 로 표준 마이크에 되돌림. autoFailed=true 로 그 페이지에서 재시도 고리를 막음. 지금 로컬 git stash@{0} 에만 있고 푸시 안 함 — 새 세션이면 아래 diff 를 다시 적용.
  • 막힌 이유: 반영하려면 public/index.html 의 idx-vc-officemode.js?v=6 → ?v=7(한 글자, 첫 화면 예산 0바이트)이 필요하고 그 파일은 공동 금지구역. 사장님 승인 후 진행.
// idx-vc-officemode.js — TICK_MS 아래
var STALL_TICKS = 40;   // ≈1초 — 엔진이 멈춘 채 이만큼 지나면 손을 뗀다
var stallTicks = 0;

// tick() 맨 앞, `if (!on || !ctx || !anaNode || !gainNode) return;` 다음
if (ctx.state !== 'running') {
  try { if (ctx.state !== 'closed') ctx.resume(); } catch (e) {}
  if (++stallTicks >= STALL_TICKS) {
    stallTicks = 0;
    console.warn('[office] 오디오 엔진이 멈춰(' + ctx.state + ') 사무실 모드를 해제합니다');
    autoFailed = true;   // 이 페이지에서는 자동으로 다시 걸지 않는다
    try { disable(true); } catch (e) {}
  }
  return;
}
stallTicks = 0;
